# Urban Limit Reworked - province caps (gate 1). Location scope (called from location_ranks allow).
# The caps grow with the owner's institutions: see script_values/urban_limit_caps_values.txt for the table.
# (Until 2026-11-24 these were fixed hand-written blocks per province size: 1-3 locations 1 settlement,
#  4-5 locations 2, 6+ locations 3, and at most 1 city per province.)

# A new town needs a free settlement slot in the province.
can_upgrade_to_town_under_province_cap = {
	urb_free_settlement_slots > 0
}

# A town becoming a city keeps the settlement count; it needs a free city slot in the province.
can_upgrade_to_city_under_province_cap = {
	urb_free_city_slots > 0
}
