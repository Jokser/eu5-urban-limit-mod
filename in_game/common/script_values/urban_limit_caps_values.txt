# Urban Limit Reworked - province caps that grow with institutions, and the burgher gate for cities.
# Used by scripted_triggers/urban_caps_gate1_triggers_FIXED_SCOPE.txt and location_ranks (city allow).
#
# Settlements = towns + cities + megalopolises; cities = cities + megalopolises.
# Base caps by province size, then each institution the OWNER has embraced adds its own bonus
# (independent of the order they were embraced in):
#
#   locations in province:      1-3        4-5        6+
#   base                        1 (1)      2 (1)      3 (1)
#   + global_trade              -          +1         +1
#   + manufactories             +1         +0 (+1)    +0 (+1)
#   + industrialization         +0 (+1)    -          +1
#   all three                   2 (2)      3 (2)      5 (2)
#   (settlements, cities in brackets)
################################################################################

# ---- province scope: counts ----
urb_prov_settlements = {
	value = 0
	every_location_in_province = {
		limit = {
			OR = {
				location_rank = location_rank:town
				location_rank = location_rank:city
				location_rank = location_rank:megalopolis
			}
		}
		add = 1
	}
}

urb_prov_cities = {
	value = 0
	every_location_in_province = {
		limit = {
			OR = {
				location_rank = location_rank:city
				location_rank = location_rank:megalopolis
			}
		}
		add = 1
	}
}

# ---- location scope: caps for the province of this location, from its owner's institutions ----
urb_cap_settlements = {
	value = 1
	if = {
		limit = { province = { any_location_in_province = { always = yes count >= 6 } } }
		value = 3
		if = {
			limit = { owner ?= { has_embraced_institution = institution:global_trade } }
			add = 1
		}
		if = {
			limit = { owner ?= { has_embraced_institution = institution:industrialization } }
			add = 1
		}
	}
	else_if = {
		limit = { province = { any_location_in_province = { always = yes count >= 4 } } }
		value = 2
		if = {
			limit = { owner ?= { has_embraced_institution = institution:global_trade } }
			add = 1
		}
	}
	else = {
		if = {
			limit = { owner ?= { has_embraced_institution = institution:manufactories } }
			add = 1
		}
	}
}

urb_cap_cities = {
	value = 1
	if = {
		limit = { province = { any_location_in_province = { always = yes count >= 4 } } }
		if = {
			limit = { owner ?= { has_embraced_institution = institution:manufactories } }
			add = 1
		}
	}
	else = {
		if = {
			limit = { owner ?= { has_embraced_institution = institution:industrialization } }
			add = 1
		}
	}
}

urb_free_settlement_slots = {
	value = urb_cap_settlements
	subtract = province.urb_prov_settlements
}

urb_free_city_slots = {
	value = urb_cap_cities
	subtract = province.urb_prov_cities
}

# ---- location scope: burghers living here (pop sizes are in thousands) ----
urb_location_burghers = {
	value = 0
	every_pop = {
		limit = { pop_type = pop_type:burghers }
		add = pop_size
	}
}
